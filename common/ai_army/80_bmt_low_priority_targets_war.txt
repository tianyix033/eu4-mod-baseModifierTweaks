# Targets not worth marching to: too far away, or an isolated island.
#
# The only rule here that ADDS. The score is a cost, so +20000 makes these
# provinces expensive to march to - comparable to the engine's own
# 'wrong region' penalty of +9000 to +20000.
#
# This used to be a multiplier, and that was the whole problem. Every additive
# term in the score is a penalty - over supply limit, not focal province,
# distance - so multiplying scaled the penalties, and a negative multiplier
# turned the entire penalty stack into a reward. The worse a province was, the
# more the army wanted it. Addition cannot invert anything.
#
# Both rules share this file so their penalties stay in one place.
#
# Distant: an enemy province in neither my region nor my continent. Colonial
# nations are exempt, since expansion far from the overlord is their whole
# purpose.
#
# Isolated island: note is_in_capital_area is defined relative to the
# province's OWNER - "connected to the capital of its owner" - so this reads as
# an overseas scrap its own ruler cannot easily reinforce. Exactly the sort of
# thing an AI loads a stack onto boats for while losing the real war at home.

province = {
	war = {
		active = {
			is_at_war = yes
		}
		eval_add = {
			factor = 0
			modifier = {
				factor = 20000
				OR = {
					AND = {
						owner = { war_with = ROOT }
						NOT = { owned_by = ROOT }
						NOT = {
							AND = {
								OR = {
									region = ROOT
									continent = ROOT
								}
								ROOT = { is_colonial_nation = no }
							}
						}
					}
					AND = {
						island = yes
						is_in_capital_area = no
					}
				}
			}
		}
	}
}
