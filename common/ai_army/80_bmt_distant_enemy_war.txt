# An enemy province in neither my region nor my continent.
#
# The Qing-marching-through-Tibet-to-besiege-Mongolia file. Negative factor, so
# these provinces sink to the bottom of the list rather than merely ranking
# lower.
#
# Colonial nations are exempt: their whole existence is expansion far from the
# overlord's region, and penalising that would freeze them in place.
#
# This overlaps with BORDER_DISTANCE_SCORE_IMPACT, which we raised from 3.0 to
# 4.0. The define is a smooth gradient on distance; this is a hard cliff at the
# region and continent boundary. If the AI turns out to be too reluctant to
# campaign abroad, weaken this before weakening the define - a cliff is easier
# to reason about than a curve.

province = {
	war = {
		active = {
			is_at_war = yes
		}
		eval_multiply = {
			factor = -0.8
			modifier = {
				owner = { war_with = ROOT }
				NOT = { owned_by = ROOT }
				NOT = {
					AND = {
						OR = {
							region = ROOT
							continent = ROOT
						}
						ROOT = { is_colonial_nation = no }
					}
				}
			}
		}
	}
}
