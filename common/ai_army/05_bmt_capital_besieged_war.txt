# My capital is threatened and I have nobody standing in it.
#
# Strongest pull in the directory at -20000. The score is a COST and lowest
# wins, so we SUBTRACT to attract. An AI that lets its capital fall while it
# counter-sieges someone else's backwater is the behaviour this whole set
# exists to stop.

province = {
	war = {
		active = {
			is_at_war = yes
		}
		eval_add = {
			factor = 0
			modifier = {
				factor = -20000
				owned_by = ROOT
				is_capital = yes
				OR = {
					NOT = {
						OR = {
							sieged_by = ROOT
							controlled_by = ROOT
						}
					}
					AND = {
						has_siege = yes
						NOT = { units_in_province = ROOT }
					}
				}
			}
		}
	}
}
