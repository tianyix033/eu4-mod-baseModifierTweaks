# BaseModifiersTweaks
#
# Governing capacity used to be a cliff: nothing at all until you crossed 100%,
# then over_governing_capacity_modifier all at once. This is the shallow first
# step, so a state that is nearly full already feels it.
#
# potential is the cheap pre-filter and trigger is the real condition, the same
# split MandateOfHeavenTweaks uses for lost_control_of_xian.
#
# Values are deliberately about a quarter of the over-capacity ones, and carry
# the same kinds of penalty so the two read as one escalating problem rather
# than two unrelated ones. Nothing diplomatic here: overreach is an internal
# failure, not something foreign courts should punish you for.
#
# A new file rather than a copy of vanilla's 00_triggered_modifiers.txt - the
# directory concatenates, so there is no need to carry 2000 lines to add one
# block. No @ markers: nothing here is copied.

	potential = {
		governing_capacity_percentage = 0.7
	}

		governing_capacity_percentage = 0.75
	}

	advisor_cost = 0.15

	yearly_absolutism = -0.25
	yearly_revolutionary_zeal = -0.25
	idea_cost = 0.05
	reform_progress_growth = -0.1
	global_rebel_suppression_efficiency = -0.1
}
